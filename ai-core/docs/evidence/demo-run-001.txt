def is_leap_year(year: int) -> bool:
    """Retorna True se `year` for bissexto no calendário gregoriano.

    Regras: divisível por 4, exceto séculos; séculos só são bissextos
    se divisíveis por 400. Entradas que não sejam int (incluindo bool)
    levantam TypeError.
    """
    if type(year) is not int:
        raise TypeError(f"year must be int, got {type(year).__name__}")
    return year % 4 == 0 and (year % 100 != 0 or year % 400 == 0)

import pytest

from leap import is_leap_year


def test_basic_leap():
    assert is_leap_year(2024) is True


def test_basic_common():
    assert is_leap_year(2023) is False


@pytest.mark.parametrize("year", [1900, 2100, 1800, 2200])

    assert is_leap_year(year) is False

def test_century_divisible_by_400_is_leap(year):
    assert is_leap_year(year) is True

    assert is_leap_year(year) is True


@pytest.mark.parametrize("year", [2019, 2021, 2022, 1901])
def test_not_divisible_by_4_is_common(year):
    assert is_leap_year(year) is False
